# Высота: три числа в translation = X Y Z
#   X — влево(-)/вправо(+) от точки привязки
#   Y — ВЫСОТА: вниз(-)/вверх(+)
#   Z — назад(-)/вперёд(+)
#
# Пример: -3.5  -0.56  0.05
#            X      Y     Z

/dh displays attribute don translation 0 -4.73 0
/dh displays attribute don translation 0 -5.20 0
